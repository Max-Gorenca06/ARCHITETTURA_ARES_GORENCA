    int count = 0;
    for (int i = 0; i < size; i++) {

            count++;
        }
    }
    return count;

 */
.globl _start
.data
    arr:   .word 12, 4, 7, 20, 3, 15, 1    # Array di test
    size:  .word 7                         # Dimensione dell'array
    k:     .word 10                        # Valore soglia K (deve contare i minori
    
.text
_start:
    la   a0, arr              # a0 = indirizzo base dell'array
    la   t0, size
    lw   a1, 0(t0)            # a1 = size (7)
    la   t0, k
    lw   a2, 0(t0)            # a2 = k (10)
    
    jal  ra, count_less       # Chiamata alla procedura
    
    # Fine programma (Exit ecall)
    li   a7, 10
    ecall

count_less:

    li t0,0 #i=0
    li t1,0 #count
    
    loop:
    bge t0,a1,end_loop #i>size

    #carico l'offset
    slli t2,t0,2 #t2=i*4->OFFSET
    #calcolo arr[i]
    add t3,a0,t2 # t3=ind  base + offset
    lw t4,0(t3) #t4=arr[i]

    bge t4,a2,salto_incr #controllo se maggiore 
    addi t1,t1,1
    salto_incr:
    addi t0,t0,1
    j loop

    end_loop:
        mv a0,t1
        ret
